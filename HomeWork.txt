(short, short) CalculationSumMultiplyArray(short[] array)
{
    short sum = 0;
    short product = 1;
    foreach (short number in array)
    {
        sum += number;
        product *= number;
    }
    return (sum, product);
}

var (sum, product) = CalculationSumMultiplyArray(numbers);
Console.WriteLine($"Sum: {sum}, Product: {product}");

short MaxValueInNumbers(short[] numbers)
{
    short max = numbers[0];
    foreach (short number in numbers)
    {
        if (number > max)
        {
            max = number;
        }
    }
    return max;
}
short MinValueInNumbers(short[] numbers)
{
    short min = numbers[0];
    foreach (short number in numbers)
    {
        if (number < min)
        {
            min = number;
        }
    }
    return min;
}

short max = MaxValueInNumbers(numbers);
short min = MinValueInNumbers(numbers);
Console.WriteLine($"The maximum element in the numbers is: {max}");
Console.WriteLine($"The minimum element in the numbers is: {min}");
void ChangeMinMaxInNumbers(short[] numbers, short min, short max)
{
    for (byte i = 0; i < numbers.Length; i++)
    {
        if (numbers[i] == min)
        {
            numbers[i] = max;
        }
        else if (numbers[i] == max)
        {
            numbers[i] = min;
        }
        Console.WriteLine($"El: {numbers[i]}");
    }
}
ChangeMinMaxInNumbers(numbers, min, max);

void FindIndexFirstPositiveOrNegativeElemInNumbers(short[] numbers, string userInput = "True")
{
    if (userInput.Equals("False", StringComparison.OrdinalIgnoreCase))
    {
        for (byte i = 0; i < numbers.Length; i++)
        {
            if (numbers[i] < 0)
            {
                Console.WriteLine($"Index of first negative element: {i}");
                return;
            }
        }
        Console.WriteLine("No negative element found.");
    }
    else
    {
        for (byte i = 0; i < numbers.Length; i++)
        {
            if (numbers[i] >= 0)
            {
                Console.WriteLine($"Index of first positive element: {i}");
                return;
            }
        }
        Console.WriteLine("No positive element found.");
    }  
}
